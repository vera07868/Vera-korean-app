// === 優化版：智慧挑選高音質語音庫 === function initSpeechVoices() { if (!("speechSynthesis" in window)) return; const findVoice = () => { const voices = window.speechSynthesis.getVoices(); if (!voices.length) return; const koVoices = voices.filter(v => v.lang === "ko-KR" || v.lang.startsWith("ko")); // 優先挑選自然、高品質的發音人（如 Edge 的 SunHi/InJoon、Google 韓語、Apple Yuna/Sora） cachedKoreanVoice = koVoices.find(v => /natural|online|google|yuna|sora|seoyeon|enhanced|premium/i.test(v.name)) || koVoices[0] || null; }; findVoice(); if (window.speechSynthesis.onvoiceschanged !== undefined) { window.speechSynthesis.onvoiceschanged = findVoice; } } window.speak = function(text) { const cleanWord = String(text || "").trim(); if (!cleanWord || !("speechSynthesis" in window)) return; window.speechSynthesis.cancel(); if (window.speechSynthesis.paused) { window.speechSynthesis.resume(); } const rate = getSpeechRate(); const utter = new SpeechSynthesisUtterance(cleanWord); utter.lang = "ko-KR"; if (!cachedKoreanVoice) { initSpeechVoices(); } if (cachedKoreanVoice) { utter.voice = cachedKoreanVoice; } // 稍微將音高調為 1.05，韓語的緊音和尾音語調會自然很多 utter.rate = rate; utter.pitch = 1.05; utter.onerror = () => { window.speechSynthesis.cancel(); }; window.speechSynthesis.speak(utter); };
